Table users {
  id uuid [pk, not null, unique]
  first_name varchar
  last_name varchar
  avatar blob
  email_address varchar [not null, unique]
  password_digest varchar [not null]
  status user_status [default: 0]
  birthdate date
  created_at timestamp [not null]
  updated_at timestamp [not null]
}

Table sessions {
  id uuid [pk, not null, unique]
  user_id uuid [not null, ref: > users.id]
  ip_address varchar
  user_agent varchar
  created_at timestamp [not null]
  updated_at timestamp [not null]
}

Table categories {
  id uuid [pk, not null, unique]
  name varchar [not null, unique]
  color varchar [not null, default: "#FFFFFF"]
  slug varchar [not null, unique]
  position int [not null, default: 1]
  created_at timestamp [not null]
  updated_at timestamp [not null]
} // Rock, Metal, Classic, Country

Table playlists {
  id uuid [pk, not null, unique]
  image blob
  name varchar [not null]
  position int [default: 1]
  status playlist_status [default: 0]
  slug varchar [not null, unique]
  user_id uuid [not null, ref: > users.id]
  created_at timestamp [not null]
  updated_at timestamp [not null]
} // unique by (name, user_id); position 0 is auto-created "Saved Songs"

Table playlist_songs {
  id uuid [pk, not null, unique]
  playlist_id uuid [not null, ref: > playlists.id]
  song_id uuid [not null, ref: > songs.id]
  position int [default: 1]
  added_at timestamp
  created_at timestamp [not null]
  updated_at timestamp [not null]
} // many playlists can have many songs

Table songs {
  id uuid [pk, not null, unique]
  image blob
  audio blob
  name varchar [not null]
  category_id uuid [not null, ref: > categories.id]
  album_id uuid [not null, ref: > albums.id]
  lyrics text
  duration_ms int
  age int
  slug varchar [not null, unique]
  monthly_listeners int [default: 0]
  position int [default: 1]
  popular boolean [default: false]
  created_at timestamp [not null]
  updated_at timestamp [not null]
} // unique by (name, album_id) — Aerials

Table albums {
  id uuid [pk, not null, unique]
  image blob
  name varchar [not null]
  slug varchar [not null, unique]
  category_id uuid [not null, ref: > categories.id]
  author_id uuid [not null, ref: > authors.id]
  release_date date
  position int [default: 1]
  created_at timestamp [not null]
  updated_at timestamp [not null]
} // unique by (name, author_id) — Toxicity

Table song_authors {
  id uuid [pk, not null, unique]
  song_id uuid [not null, ref: > songs.id]
  author_id uuid [not null, ref: > authors.id]
  position int [default: 1]
  created_at timestamp [not null]
  updated_at timestamp [not null]
}

Table authors {
  id uuid [pk, not null, unique]
  image blob
  name varchar [not null, unique]
  description text
  slug varchar [not null, unique]
  user_id uuid [ref: > users.id]
  created_at timestamp [not null]
  updated_at timestamp [not null]
} // System of a Down

Table popular_songs {
  id uuid [pk, not null, unique]
  author_id uuid [not null, ref: > authors.id]
  song_id uuid [not null, ref: > songs.id]
  position int [default: 1]
  created_at timestamp [not null]
  updated_at timestamp [not null]
} // unique by (author_id, song_id)

Table song_queues {
  id uuid [pk, not null, unique]
  user_id uuid [not null, ref: > users.id]
  song_id uuid [not null, ref: > songs.id]
  position int [default: 1]
  source varchar // album, popular, artist_shuffle, default
  created_at timestamp [not null]
  updated_at timestamp [not null]
}

Table play_histories {
  id uuid [pk, not null, unique]
  user_id uuid [not null, ref: > users.id]
  song_id uuid [not null, ref: > songs.id]
  played_at timestamp
  source varchar // search, or null
  created_at timestamp [not null]
  updated_at timestamp [not null]
}

Enum user_status {
  active
  blocked
  admin
}

Enum playlist_status {
  private
  public
}
